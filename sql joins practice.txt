/* 11. Write a query to find the name and numbers of all salesmen who had more than one customer. */

       name
  FROM salesman
  WHERE salesman_id IN (SELECT salesman_id
                          FROM customer
                          GROUP BY salesman_id
                          HAVING COUNT(*) > 1);
